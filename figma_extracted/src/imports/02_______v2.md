## 2. 데이터 수집 및 전처리

> 정보나루·국가서지 LOD는 무료 공공 API. 솔로몬은 공개 REST API가 없는 사서 로그인 기반 웹 분석 서비스이므로, 사서가 내보낸 장서 분석 CSV를 업로드받아 연계한다(선택적 보완 입력).

### 2-1. 도서관 정보나루 Open API

**엔드포인트**
```
GET https://data4library.kr/api/loanItemSrch
```

**주요 요청 파라미터**

| 파라미터 | 설명 | 예시 |
|----------|------|------|
| `authKey` | 발급받은 인증키 | `your_api_key` |
| `region` | 지역 코드 | `11` (서울) |
| `age` | 연령대 | `20` (20대) |
| `startDt` / `endDt` | 조회 기간 | 최근 7일 |
| `format` | 응답 포맷 | `json` |

```python
import requests
from datetime import datetime, timedelta

def get_last_monday() -> str:
    today = datetime.today()
    last_monday = today - timedelta(days=today.weekday())
    return last_monday.strftime("%Y-%m-%d")

def get_today() -> str:
    return datetime.today().strftime("%Y-%m-%d")

def fetch_popular_books(region_code: str, age: int, top_n: int = 10) -> list:
    """지역·연령 인기 대출 도서(트렌드 컨텍스트). loanItemSrch 응답에는 loan_count도 포함된다."""
    url = "https://data4library.kr/api/loanItemSrch"
    params = {
        "authKey": LIBRARY_API_KEY,
        "region": region_code,
        "age": age,
        "startDt": get_last_monday(),
        "endDt": get_today(),
        "format": "json"
    }
    response = requests.get(url, params=params)
    data = response.json()
    books = data["response"]["docs"][:top_n]
    return [
        {
            "isbn": b["doc"]["isbn13"],
            "title": b["doc"]["bookname"],
            "loan_count": int(b["doc"].get("loan_count", 0) or 0),
        }
        for b in books
    ]
```

### 2-2. 장서 회전율 산출 및 숨은 명저 선정

> **회전율은 도서관 장서의 소장·대출 건수로 직접 산출**한다. 이때 선정 대상 풀은
> 인기 대출 Top N이 아니라 **도서관의 전체 장서(대출이 저조한 도서 포함)**여야 한다.
> 인기 대출 데이터(`fetch_popular_books`)는 '숨은 명저' 판정을 돕는 **신호**로만 쓰고,
> 솔로몬 리포트 CSV(장서 평가)는 공개 API가 없어 사서 업로드 기반 **선택적 입력**으로 보강한다.

```python
import csv

def fetch_library_collection(library_code: str, page_size: int = 100) -> list:
    """
    특정 도서관의 소장 장서 + 대출 건수를 조회 (itemSrchByLib).
    회전율 = 대출 건수 / 소장 권수 산출의 기준 데이터.
    ※ 실제 파라미터/응답 필드명은 정보나루 itemSrchByLib 문서 기준으로 맞춘다.
    """
    url = "https://data4library.kr/api/itemSrchByLib"
    params = {
        "authKey": LIBRARY_API_KEY,
        "libCode": library_code,
        "pageSize": page_size,
        "format": "json",
    }
    docs = requests.get(url, params=params).json()["response"]["docs"]
    collection = []
    for d in docs:
        doc = d["doc"]
        collection.append({
            "isbn": doc.get("isbn13", ""),
            "title": doc.get("bookname", ""),
            "loan_count": int(doc.get("loan_count", 0) or 0),
            "stock_count": int(doc.get("reg_count", 1) or 1),  # 소장(등록) 권수
        })
    return collection


def load_solomon_report(csv_path: str) -> dict:
    """사서가 솔로몬에서 내보낸 장서 분석 CSV를 dict로 로드 (선택적)."""
    report = {}
    with open(csv_path, encoding="utf-8") as f:
        for row in csv.DictReader(f):
            report[row["isbn"]] = {
                "evaluation": float(row.get("evaluation", 0) or 0),  # 장서 평가 점수(0~5)
            }
    return report


def select_target_books(
    collection: list,
    solomon_report: dict | None = None,
    popular_isbns: set | None = None,
    top_n: int = 5,
) -> list:
    """
    회전율 = 대출 건수 / 소장 권수 (도서관 장서 데이터로 직접 산출).
    저회전(<0.3) 도서 중에서 제작 대상을 선정한다.

    타입 A '숨은 명저': 솔로몬 평가 높음(>=4.0) 또는 지역 인기 도서인데 회전율은 저조
    타입 B '회전율 개선': 그 외 저회전 도서
    """
    report = solomon_report or {}
    popular_isbns = popular_isbns or set()
    targets = []
    for book in collection:
        stock_count = book.get("stock_count", 1) or 1
        turnover_rate = book.get("loan_count", 0) / stock_count
        if turnover_rate >= 0.3:
            continue

        evaluation = report.get(book["isbn"], {}).get("evaluation", 0)
        is_hidden_gem = evaluation >= 4.0 or book["isbn"] in popular_isbns
        book["type"] = "A" if is_hidden_gem else "B"
        book["turnover_rate"] = turnover_rate
        targets.append(book)

    return sorted(targets, key=lambda x: x["turnover_rate"])[:top_n]
```

### 2-2-1. 쇼츠·SNS 화제 도서 선정 (타입 C)

> 유튜브 쇼츠·릴스·북톡에서 화제가 된 책도 제작 대상에 포함한다. 다만 SNS 화제도를
> 직접 받는 무료 공개 API는 없으므로, **두 가지 현실적 신호**로 '뜨는 책'을 포착한다.
> ① 정보나루 **대출 급상승(`hotTrend`)** — 전주 대비 대출 상승폭이 큰 도서(트렌드 프록시).
> ② 사서가 입력하는 **트렌드 큐레이션 CSV** — 북튜브·북톡 화제작을 사람이 직접 반영.
> 화제 도서는 회전율과 무관하게 포함하며, 우리 도서관에 없더라도 `locate`로 **가까운
> 소장 도서관**을 안내해 온라인 관심을 오프라인 대출로 전환한다.

```python
def fetch_trending_books(region_code: str, age: int, top_n: int = 5) -> list:
    """
    정보나루 대출 급상승(hotTrend)으로 '지금 뜨는 화제 도서' 포착.
    ※ 실제 파라미터/응답 필드명은 정보나루 hotTrend 문서 기준으로 맞춘다.
    """
    url = "https://data4library.kr/api/hotTrend"
    params = {
        "authKey": LIBRARY_API_KEY,
        "searchDt": get_last_monday(),
        "format": "json",
    }
    results = requests.get(url, params=params).json()["response"]["results"]
    trending = []
    for r in results[:top_n]:
        doc = r["result"]["docs"][0]["doc"]
        trending.append({
            "isbn": doc.get("isbn13", ""),
            "title": doc.get("bookname", ""),
            "trend_rank": int(doc.get("ranking", 0) or 0),
            "source": "loan_surge",
        })
    return trending


def load_trend_curation(csv_path: str) -> list:
    """사서가 입력한 북튜브·북톡 화제작 리스트 CSV (isbn,title 컬럼)."""
    items = []
    with open(csv_path, encoding="utf-8") as f:
        for row in csv.DictReader(f):
            items.append({
                "isbn": row["isbn"].strip(),
                "title": row.get("title", "").strip(),
                "source": "curation",
            })
    return items


def build_weekly_targets(
    collection: list,
    region_code: str,
    age: int,
    solomon_report: dict | None = None,
    popular_isbns: set | None = None,
    trend_csv_path: str | None = None,
    ab_top_n: int = 5,
) -> list:
    """타입 A/B(숨은 명저·회전율 개선) + 타입 C(쇼츠·SNS 화제 도서)를 합쳐 주간 제작 목록 구성."""
    ab = select_target_books(collection, solomon_report, popular_isbns, top_n=ab_top_n)

    trending = fetch_trending_books(region_code, age)
    if trend_csv_path:
        trending += load_trend_curation(trend_csv_path)

    seen = {b["isbn"] for b in ab}
    for t in trending:
        if not t["isbn"] or t["isbn"] in seen:
            continue  # A/B에 이미 잡힌 책은 중복 제외
        t["type"] = "C"
        ab.append(t)
        seen.add(t["isbn"])

    return ab
```

> 타입 C 도서는 회전율 필터를 거치지 않으므로 우리 도서관 장서에 없을 수 있다.
> 이 경우에도 영상은 제작되고, QR 랜딩의 `locate`(아래 5-2)가 `libSrchByBook`으로
> **해당 ISBN을 소장한 가장 가까운 도서관**을 찾아 안내한다.

### 2-3. 국가서지 LOD SPARQL 조회

> **✅ 라이브 엔드포인트로 검증된 술어** (`poc/pipeline.py`로 실측). 국립중앙도서관 LOD는
> schema.org가 아니라 `bibo`·`dcterms`·`dc` 표준 어휘를 혼용하며, 아래가 실제 동작하는 술어다.
>
> | 항목 | 술어 | 비고 |
> |------|------|------|
> | ISBN | `bibo:isbn` | **하이픈 없는 ISBN-13 문자열**로 매칭 (`isbn13`/`schema:isbn`은 0건) |
> | 제목 | `dcterms:title` | |
> | 저자명 | `dc:creator` (DC elements 1.1) | `dcterms:creator`는 저자 **URI**를 반환 → 이름은 `dc:creator` |
> | 주제어 | `dcterms:subject` → `rdfs:label` | 주제는 KSH 자원 URI라 라벨을 한 번 더 조회 |
>
> 또한 LOD에는 '목차/줄거리'가 없으므로 대본 입력은 제목·저자·주제분류로 구성한다.
> NLK 엔진(ontobase)은 ① 리터럴 끝에 `~`를 덧붙이고 ② `<subj> ?p ?o` 형태 쿼리에서
> 오류를 내므로, 술어를 명시한 패턴으로 조회하고 값 끝의 `~`를 제거한다.

```python
from SPARQLWrapper import SPARQLWrapper, JSON

NLK_SPARQL_ENDPOINT = "https://lod.nl.go.kr/sparql"

def _clean(v: str) -> str:
    return v.rstrip("~").strip() if v else v  # ontobase 엔진이 붙이는 '~' 제거

def fetch_bibliographic_data(isbn: str) -> dict:
    sparql = SPARQLWrapper(NLK_SPARQL_ENDPOINT)
    sparql.setQuery(f"""
        PREFIX bibo:    <http://purl.org/ontology/bibo/>
        PREFIX dcterms: <http://purl.org/dc/terms/>
        PREFIX dc:      <http://purl.org/dc/elements/1.1/>
        PREFIX rdfs:    <http://www.w3.org/2000/01/rdf-schema#>
        SELECT ?title ?author ?subjectLabel
        WHERE {{
          ?work bibo:isbn "{isbn}" .
          OPTIONAL {{ ?work dcterms:title   ?title  . }}
          OPTIONAL {{ ?work dc:creator      ?author . }}
          OPTIONAL {{ ?work dcterms:subject ?s . ?s rdfs:label ?subjectLabel . }}
        }}
        LIMIT 1
    """)
    sparql.setReturnFormat(JSON)
    sparql.setTimeout(40)
    bindings = sparql.query().convert()["results"]["bindings"]
    if not bindings:
        return {"isbn": isbn, "title": "", "author": "", "subject": ""}

    row = bindings[0]
    return {
        "isbn":    isbn,
        "title":   _clean(row.get("title",        {}).get("value", "")),
        "author":  _clean(row.get("author",       {}).get("value", "")),
        "subject": _clean(row.get("subjectLabel", {}).get("value", "")),
    }
```

> 실측 예: `fetch_bibliographic_data("9788937473135")` →
> `{title: "82년생 김지영", author: "조남주", subject: "한국 현대 소설"}`.
> 동작하는 단일 ISBN PoC 전체 파이프라인은 [`poc/`](../poc/) 참고.

> `client.models.generate_content(model=..., ...)` 형태로 호출한다.

### 3-2. 구현 코드

```python
from google import genai
from google.genai import types
import os

client = genai.Client(api_key=os.environ["GEMINI_API_KEY"])

SYSTEM_PROMPT = """
당신은 유튜브 쇼츠 및 인스타그램 릴스 전문 카피라이터입니다.
도서관 사서가 제공한 책 정보를 바탕으로 1분 분량(약 150~180자)의
숏폼 북트레일러 대본을 작성합니다.

규칙:
1. 첫 문장은 반드시 시청자가 멈춰서 볼 만한 질문 또는 충격적 상황 제시로 시작
2. 결말 및 핵심 반전은 절대 스포일러하지 않음
3. 마지막 문장은 "뒷이야기가 궁금하다면 지금 도서관에서 빌려보세요"로 마무리
4. 전체 톤은 책의 장르에 맞게 조정 (스릴러: 긴박감 / 로맨스: 설렘 / 에세이: 공감)
5. 대본만 출력하고 부가 설명은 쓰지 않음
"""

def generate_script(book_info: dict) -> str:
    # LOD에는 목차/줄거리가 없을 수 있으므로 제목·저자·주제분류만 입력으로 쓴다(모두 .get 안전 접근).
    user_prompt = f"""
    책 제목: {book_info.get('title', '')}
    저자: {book_info.get('author', '')}
    주제 분류: {book_info.get('subject', '')}

    위 정보를 바탕으로 숏폼 북트레일러 대본을 작성해주세요.
    """

    response = client.models.generate_content(
        model="gemini-2.5-flash",
        contents=user_prompt,
        config=types.GenerateContentConfig(system_instruction=SYSTEM_PROMPT),
    )
    return response.text.strip()
```

### 3-3. 스크립트 출력 예시


### 4-1. TTS 음성 합성 — Edge TTS (완전 무료)

API 키 없이 사용 가능한 Microsoft Edge TTS입니다.
한국어 음성 품질이 유료 서비스에 준하는 수준입니다.

**설치**
```bash
pip install edge-tts
```

**장르별 음성 설정**
```python
import edge_tts
import asyncio

# 주의: edge-tts의 rate/pitch는 반드시 부호를 포함해야 한다 ("+0%", "+0Hz"). 부호가 없으면 오류.
GENRE_VOICE_MAP = {
    "스릴러":  {"voice": "ko-KR-InJoonNeural",  "rate": "+10%", "pitch": "-5Hz"},  # 남성, 긴박
    "로맨스":  {"voice": "ko-KR-SunHiNeural",   "rate": "-5%",  "pitch": "+3Hz"},  # 여성, 따뜻
    "에세이":  {"voice": "ko-KR-SunHiNeural",   "rate": "-10%", "pitch": "+0Hz"},  # 여성, 차분
    "판타지":  {"voice": "ko-KR-InJoonNeural",  "rate": "+0%",  "pitch": "+5Hz"},  # 남성, 웅장
    "default": {"voice": "ko-KR-SunHiNeural",   "rate": "+0%",  "pitch": "+0Hz"},
}

async def generate_tts_audio(script: str, genre: str, output_path: str) -> str:
    config = GENRE_VOICE_MAP.get(genre, GENRE_VOICE_MAP["default"])
    communicate = edge_tts.Communicate(
        text=script,
        voice=config["voice"],
        rate=config["rate"],
        pitch=config["pitch"]
    )
    await communicate.save(output_path)
    return output_path

# 동기 래퍼 — 스케줄러 등 '동기' 컨텍스트 전용.
# FastAPI async 라우트 내부에서는 asyncio.run()이 "running event loop" 오류를 내므로
# 라우트에서는 await generate_tts_audio(...)를 직접 호출할 것.
def tts_sync(script: str, genre: str, output_path: str) -> str:
    return asyncio.run(generate_tts_audio(script, genre, output_path))
```

### 4-2. 배경 이미지 — Pexels API (완전 무료)

DALL·E 3 대신 저작권 무료 고화질 스톡 이미지를 활용합니다.
상업적 이용 포함 완전 무료이며 API 키는 https://www.pexels.com/api/ 에서 무료 발급합니다.

```python
import requests
import os

PEXELS_API_KEY = os.environ["PEXELS_API_KEY"]

GENRE_IMAGE_QUERY = {
    "스릴러":  "dark mystery night city",
    "로맨스":  "soft warm light couple",
    "에세이":  "cozy cafe book reading",
    "판타지":  "magical forest fantasy light",
    "인문":    "library books knowledge",
    "default": "open book dramatic light",
}

def fetch_background_image(genre: str, output_path: str) -> str:
    query = GENRE_IMAGE_QUERY.get(genre, GENRE_IMAGE_QUERY["default"])

    response = requests.get(
        "https://api.pexels.com/v1/search",
        headers={"Authorization": PEXELS_API_KEY},
        params={
            "query": query,
            "orientation": "portrait",  # 9:16 세로 포맷
            "size": "large",
            "per_page": 5
        }
    )
    photos = response.json()["photos"]
    if not photos:
        raise ValueError(f"Pexels 검색 결과 없음: {query}")

    image_url = photos[0]["src"]["portrait"]
    image_data = requests.get(image_url).content
    with open(output_path, "wb") as f:
        f.write(image_data)
    return output_path
```

> **대안:** 국가서지 LOD에 책 표지 이미지 URL이 포함된 경우 책 표지를 배경으로 직접 사용하면 더욱 직관적입니다.

```python
import subprocess

def render_video(audio_path: str, image_path: str, script: str,
                 output_path: str, qr_path: str | None = None) -> str:
    srt_path = generate_srt_from_script(script, audio_path)
    duration = get_audio_duration(audio_path)

    style = (
        f"subtitles={srt_path}:force_style='"
        "FontName=NanumGothicBold,"
        "FontSize=22,"
        "PrimaryColour=&HFFFFFF,"
        "OutlineColour=&H000000,"
        "Outline=2,"
        "Alignment=2'"
    )
    base_vf = (
        "scale=1080:1920:force_original_aspect_ratio=increase,"
        "crop=1080:1920,"
        f"{style}"
    )

    cmd = ["ffmpeg", "-y", "-loop", "1", "-i", image_path, "-i", audio_path]
    if qr_path:
        # QR을 영상 마지막 3초간 우하단에 오버레이 (overlay 필터)
        cmd += ["-i", qr_path]
        filter_complex = (
            f"[0:v]{base_vf}[bg];"
            "[2:v]scale=240:240[qr];"
            f"[bg][qr]overlay=W-w-60:H-h-180:enable='gte(t,{duration - 3:.2f})'[v]"
        )
        cmd += ["-filter_complex", filter_complex, "-map", "[v]", "-map", "1:a"]
    else:
        cmd += ["-vf", base_vf]

    cmd += [
        "-c:v", "libx264",
        "-tune", "stillimage",
        "-c:a", "aac",
        "-b:a", "192k",
        "-pix_fmt", "yuv420p",
        "-shortest",
        output_path,
    ]

    subprocess.run(cmd, check=True)
    return output_path


def generate_srt_from_script(script: str, audio_path: str) -> str:
    """
    실제 오디오 길이(ffprobe)에 맞춰 문장 단위 SRT 자막 생성.
    TTS 길이는 가변적이므로 60초 하드코딩 대신 실제 길이로 균등 분배해야
    자막이 음성/영상과 어긋나지 않는다.
    정밀 싱크가 필요하면 faster-whisper 단어 단위 정렬로 대체 가능 (로컬, 무료)
    """
    sentences = [s.strip() for s in script.split("\n") if s.strip()]
    total_duration = get_audio_duration(audio_path)
    total_chars = sum(len(s) for s in sentences) or 1

    srt_lines = []
    cursor = 0.0
    for i, sentence in enumerate(sentences):
        # 문장 길이(글자 수) 비율로 가중 분배 → 짧은/긴 문장 읽히는 속도와 맞춤
        seg = total_duration * (len(sentence) / total_chars)
        start, end = cursor, cursor + seg
        cursor = end
        srt_lines += [str(i + 1), f"{fmt_time(start)} --> {fmt_time(end)}", sentence, ""]

    srt_path = "/tmp/subtitle.srt"
    with open(srt_path, "w", encoding="utf-8") as f:
        f.write("\n".join(srt_lines))
    return srt_path


def get_audio_duration(audio_path: str) -> float:
    """ffprobe로 오디오 파일의 실제 재생 길이(초)를 반환."""
    result = subprocess.run(
        ["ffprobe", "-v", "error", "-show_entries", "format=duration",
         "-of", "default=noprint_wrappers=1:nokey=1", audio_path],
        capture_output=True, text=True, check=True,
    )
    return float(result.stdout.strip())


def fmt_time(seconds: float) -> str:
    h = int(seconds // 3600)
    m = int((seconds % 3600) // 60)
    s = int(seconds % 60)
    ms = int((seconds % 1) * 1000)
    return f"{h:02}:{m:02}:{s:02},{ms:03}"
```

### 5-2. 위치 기반 도서관 안내 — Kakao Maps API (월 30만 크레딧 무료)

```python
from fastapi import FastAPI, Query
from typing import Optional
from math import radians, sin, cos, sqrt, atan2

app = FastAPI()

@app.get("/locate")
async def locate_library(
    isbn: str,
    lat: Optional[float] = Query(None),
    lng: Optional[float] = Query(None)
):
    libraries = await fetch_libraries_with_book(isbn)  # 정보나루 API 재활용

    if lat and lng:
        for lib in libraries:
            lib["distance_km"] = haversine(lat, lng, lib["lat"], lib["lng"])
        libraries = sorted(libraries, key=lambda x: x.get("distance_km", 9999))

    available = [lib for lib in libraries if lib.get("loan_available")]
    return {"isbn": isbn, "nearest_libraries": available[:3]}


def haversine(lat1, lng1, lat2, lng2) -> float:
    R = 6371
    dlat = radians(lat2 - lat1)
    dlng = radians(lng2 - lng1)
    a = sin(dlat/2)**2 + cos(radians(lat1)) * cos(radians(lat2)) * sin(dlng/2)**2
    return R * 2 * atan2(sqrt(a), sqrt(1 - a))


async def fetch_libraries_with_book(isbn: str) -> list:
    """
    정보나루 libSrchByBook: 특정 ISBN을 소장한 도서관 목록 조회.
    ※ 실제 응답 필드명은 정보나루 libSrchByBook 문서 기준으로 맞춘다.
    """
    url = "https://data4library.kr/api/libSrchByBook"
    params = {"authKey": LIBRARY_API_KEY, "isbn": isbn, "format": "json"}
    libs = requests.get(url, params=params).json()["response"]["libs"]
    return [
        {
            "name": d["lib"].get("libName"),
            "address": d["lib"].get("address"),
            "lat": float(d["lib"].get("latitude", 0) or 0),
            "lng": float(d["lib"].get("longitude", 0) or 0),
            "loan_available": True,
        }
        for d in libs
    ]
```

> `fetch_libraries_with_book`는 설명을 위해 동기 `requests`로 적었으나, async 라우트에서는
> 이벤트 루프를 막지 않도록 **`httpx.AsyncClient`** 또는 `run_in_executor`로 감싸서 호출한다.

### 5-3. 비동기 Job 처리 (FastAPI BackgroundTasks)

> `POST /api/generate`는 즉시 `job_id`를 반환하고, 실제 생성은 백그라운드에서 수행한다.
> 프로토타입은 인메모리 Job 스토어로 충분하며, 정식 서비스는 Redis/DB 또는 Celery로 확장한다.

```python
from fastapi import BackgroundTasks, HTTPException
from pydantic import BaseModel
from uuid import uuid4

JOBS: dict[str, dict] = {}  # 프로토타입: 인메모리 잡 스토어

class GenerateRequest(BaseModel):
    isbn: str
    library_code: str
    region_code: str = "11"
    age_group: int = 20


GENRE_KEYWORDS = {
    "추리": "스릴러", "스릴러": "스릴러", "로맨스": "로맨스", "연애": "로맨스",
    "에세이": "에세이", "수필": "에세이", "판타지": "판타지", "환상": "판타지",
}

def detect_genre(subject: str) -> str:
    for keyword, genre in GENRE_KEYWORDS.items():
        if keyword in (subject or ""):
            return genre
    return "default"


def run_pipeline(job_id: str, req: GenerateRequest) -> None:
    """단일 ISBN → 대본 → TTS → 이미지 → MP4 → QR 합성 파이프라인 (동기 실행)."""
    try:
        book = fetch_bibliographic_data(req.isbn)
        genre = detect_genre(book.get("subject", ""))
        script = generate_script(book)
        audio = tts_sync(script, genre, f"outputs/{job_id}.mp3")
        image = fetch_background_image(genre, f"outputs/{job_id}.jpg")
        qr = generate_qr_code(req.isbn, f"outputs/qr_{job_id}.png")
        video = render_video(audio, image, script, f"outputs/{job_id}.mp4", qr_path=qr)
        JOBS[job_id] = {"status": "completed", "result": {
            "video_path": video, "script": script, "book": book, "qr_code_path": qr,
        }}
    except Exception as e:  # 프로토타입 수준 오류 기록
        JOBS[job_id] = {"status": "failed", "error": str(e)}


@app.post("/api/generate")
async def create_job(req: GenerateRequest, background_tasks: BackgroundTasks):
    job_id = f"job_{uuid4().hex[:8]}"
    JOBS[job_id] = {"status": "processing", "result": None}
    background_tasks.add_task(run_pipeline, job_id, req)
    return {"job_id": job_id, "status": "processing", "estimated_seconds": 90}


@app.get("/api/job/{job_id}")
async def get_job(job_id: str):
    job = JOBS.get(job_id)
    if job is None:
        raise HTTPException(status_code=404, detail="job not found")
    return {"job_id": job_id, **job}
```

### 5-4. 주간 자동화 스케줄러 (APScheduler)

```python
def weekly_job() -> None:

    region_code, age, library_code = "11", 20, "111024"
```

```
2. 회전율 산출 (정보나루 소장/대출 건수)            [무료]
   └─ turnover_rate = 대출 건수 / 소장 건수
   └─ (선택) 솔로몬 내보내기 CSV 업로드 → 장서 평가로 '숨은 명저' 보강

2-1. 화제 도서 포착 (타입 C)                        [무료]
   └─ 정보나루 대출 급상승(hotTrend) + (선택) 사서 트렌드 큐레이션 CSV

3. 교차 분석 (build_weekly_targets)
   └─ 회전율 30% 미만 → 타입 A/B 선정 + 화제 도서(타입 C) 합산 → 제작 대상 ISBN

4. 국가서지 LOD SPARQL 조회 (NLK 온톨로지)         [무료]
   └─ 서지 정보(제목·저자·주제분류) 수집

5. Gemini 2.5 Flash 스크립트 생성                  [무료]
   └─ 서지 데이터 → 장르 감지 → 후킹 스크립트 (150~180자)

6. Edge TTS 오디오 생성                            [무료]
   └─ 장르별 한국어 음성 → MP3 파일

7. Pexels API 배경 이미지 수집                      [무료]
   └─ 장르 키워드 → 세로형(9:16) 고화질 이미지

8. FFmpeg 영상 렌더링 (로컬)                        [무료]
   └─ 이미지 + 오디오 + 자막(SRT) → MP4 (1080×1920)

9. QR코드 생성                                     [무료]
   └─ 위치 기반 랜딩 페이지 URL → QR 이미지

10. outputs/ 폴더 저장
    └─ 사서가 MP4 직접 다운로드 후 SNS 업로드
```
